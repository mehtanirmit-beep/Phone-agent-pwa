This passes: ${task}. Check endpoint and CORS settings.`); } } say('I can run Shortcuts by name, open Maps, search the web, draft SMS/email, start calls to numbers, save notes, and pass reminder text to a Shortcut. For general AI questions, configure a secure AI endpoint in Settings.'); } $('run').onclick = run; document.querySelectorAll('[data-cmd]').forEach((b) => b.onclick = () => { command.value = b.dataset.cmd; run(); }); $('endpoint').value = localStorage.getItem('pa_endpoint') || ''; $('save').onclick = () => { const u = $('endpoint').value.trim(); if (u && !/^https:\/\//i.test(u)) return say('Endpoint must start with https://'); if (u) localStorage.setItem('pa_endpoint', u); else localStorage.removeItem('pa_endpoint'); say('Endpoint setting saved.'); }; $('clear').onclick = () => { if (confirm('Delete locally saved notes and endpoint?')) { localStorage.removeItem('pa_endpoint'); localStorage.removeItem('pa_notes'); $('endpoint').value = ''; notes(); say('Local data cleared.'); } }; const Recognition = window.SpeechRecognition || window.webkitSpeechRecognition; if (!Recognition) { $('mic').onclick = () => say('Speech recognition is not available in this browser. You can use keyboard dictation instead.'); try { r.start(); } catch (e) { say('Could not start microphone: ' + e.message); } }; } notes(); if ('serviceWorker' in navigator) navigator.serviceWorker.register('./sw.js').catch(() => {});
